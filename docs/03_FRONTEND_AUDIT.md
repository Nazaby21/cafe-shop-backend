### 2.2 Safe Polling Without Memory Leaks for Bakong KHQR (`features/payment/hooks.ts`) (`[MUST]`)
```tsx
export function usePayment(orderId: number, options?: { pollWhilePending?: boolean }) {
  return useQuery({
    queryKey: queryKeys.orders.payment(orderId),
    queryFn: () => paymentApi.getPayment(orderId),
    enabled: Number.isFinite(orderId) && orderId > 0,
    retry: false,
    refetchInterval: (query) => {
      if (!options?.pollWhilePending) return false;
      const data = query.state.data;
      if (!data || data.status !== 'pending') return false;
      if (data.expires_at && new Date(data.expires_at).getTime() <= Date.now()) {
        return false;
      }
      return 3000; // Poll every 3 seconds only while pending & unexpired
    },
  });
}
```
Because TanStack Query manages the timer lifecycle tied to component mount/unmount and `pollWhilePending`, closing `PayDialog` or navigating away from `/orders/[id]` immediately clears the polling interval with zero memory leaks.
